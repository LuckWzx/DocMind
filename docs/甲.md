## 模块1：数据持久层

### User — 用户

```go
// User 系统用户，属于某个租户
type User struct {
    ID                  string          `json:"id"                     gorm:"type:varchar(36);primaryKey"`
    Username            string          `json:"username"               gorm:"type:varchar(100);uniqueIndex;not null"`  // 用户名，唯一
    Email               string          `json:"email"                  gorm:"type:varchar(255);uniqueIndex;not null"`  // 邮箱，唯一，用于登录
    PasswordHash        string          `json:"-"                     gorm:"type:varchar(255);not null"`               // 密码哈希，API不输出
    CreatedAt           time.Time       `json:"created_at"`             // 创建时间
    UpdatedAt           time.Time       `json:"updated_at"`             // 更新时间
```

### KnowledgeBase — 知识库（核心）

```go
// KnowledgeBase 知识库，知识条目的容器，一个租户下的一个知识域
type KnowledgeBase struct {
    ID                    string                     `json:"id"                     gorm:"type:varchar(36);primaryKey"`
    Name                  string                     `json:"name"                   gorm:"type:varchar(255);not null"` // 知识库名称
    Type                  string                     `json:"type"                   gorm:"type:varchar(32);default:'document'"` // 类型: document / faq
    Description           string                     `json:"description"            gorm:"type:text"`                  // 知识库描述
    EmbeddingModelID      string                     `json:"embedding_model_id"   gorm:"type:varchar(64)"`            // 绑定的Embedding模型ID
    SummaryModelID        string                     `json:"summary_model_id"     gorm:"type:varchar(64)"`            // 绑定的对话模型ID
    IsPinned              bool                       `json:"is_pinned"              gorm:"default:false"`              // 是否置顶
    ChunkingConfig        ChunkingConfig             `json:"chunking_config"        gorm:"type:json"`                  // 分块策略配置
    ExtractConfig         *ExtractConfig             `json:"extract_config"         gorm:"type:json"`                  // 知识抽取配置
    FAQConfig             *FAQConfig                 `json:"faq_config"             gorm:"type:json"`                  // FAQ特殊配置
    StorageProviderConfig *StorageProviderConfig     `json:"storage_provider_config" gorm:"column:storage_provider_config;type:jsonb"` // 存储提供方配置
    VectorStoreID         *string                    `json:"vector_store_id"        gorm:"type:varchar(36)"`           // 绑定的向量存储ID（nil表示使用租户默认）
    IndexingStrategy      IndexingStrategy           `json:"indexing_strategy"      gorm:"type:json"`                  // 索引策略
    CreatedAt             time.Time                  `json:"created_at"`              // 创建时间
    UpdatedAt             time.Time                  `json:"updated_at"`              // 更新时间
    DeletedAt             gorm.DeletedAt             `json:"deleted_at"               gorm:"index"`                    // 软删除
}
```

### Knowledge — 知识条目

```go
// Knowledge 知识条目，知识库中的一条知识（一份文档或一组FAQ）
type Knowledge struct {

    Title           string         `json:"title"            gorm:"type:varchar(255)"` // 条目标题（文件名）
    Description     string         `json:"description"      gorm:"type:text"`         // 条目描述

    FileURL         string         `json:"file_url"         gorm:"type:text"`                         // 原始文件URL
    FileType        string         `json:"file_type"        gorm:"type:varchar(32)"`                  // pdf / docx / md / txt / html...
    TagID           string         `json:"tag_id"           gorm:"type:varchar(36);index"`            // 标签ID
    ErrorMessage    string         `json:"error_message"    gorm:"type:text"`                         // 解析失败时的错误信息
    CreatedAt       time.Time      `json:"created_at"`
    UpdatedAt       time.Time      `json:"updated_at"`
    DeletedAt       gorm.DeletedAt `json:"deleted_at"       gorm:"index"`
```

### Chunk — 文本分块

```go
// Chunk 文本分块，知识被拆分后的最小检索单元
type Chunk struct {
    ID               string         `json:"id"                gorm:"type:varchar(36);primaryKey"`
    Content          string         `json:"content"           gorm:"type:text"`         // 分块文本内容
    ChunkIndex       int            `json:"chunk_index"       gorm:"index"`             // 在文档中的顺序号
    KnowledgeID      string         `json:"knowledge_id"      gorm:"type:varchar(36);index"` // 所属知识条目ID
    KnowledgeBaseID  string         `json:"knowledge_base_id" gorm:"type:varchar(36);index"` // 所属知识库ID
    ChunkType        string         `json:"chunk_type"        gorm:"type:varchar(32);default:'text'"` // text / image_ocr / image_caption / faq...
    ChunkStatus      int            `json:"chunk_status"      gorm:"default:0"`         // 0=默认 1=已存储 2=已索引
    ParentChunkID    string         `json:"parent_chunk_id"   gorm:"type:varchar(36);index"` // 父分块ID（父子策略中使用）
    TagID            string         `json:"tag_id"            gorm:"type:varchar(36);index"` // 标签ID
    Metadata         JSON           `json:"metadata"          gorm:"type:json"`         // 扩展元数据（灵活存储FAQ/文档额外信息）
    ContentHash      string         `json:"content_hash"      gorm:"type:varchar(64)"`  // 内容hash，用于去重/增量更新
    IsEnabled        bool           `json:"is_enabled"        gorm:"default:true"`      // 是否启用（禁用后不可检索）
    CreatedAt        time.Time      `json:"created_at"`
    UpdatedAt        time.Time      `json:"updated_at"`
    DeletedAt        gorm.DeletedAt `json:"deleted_at"        gorm:"index"`
}
```

### Model — 模型定义

```go
// Model 系统可用的AI模型配置
type Model struct {
    ID          string         `json:"id"          gorm:"type:varchar(36);primaryKey"`
    Name        string         `json:"name"        gorm:"type:varchar(255);not null"` // 模型名称（用户自定义）
    Type        string         `json:"type"        gorm:"type:varchar(32);not null"`  // 模型类型: Embedding / Rerank / KnowledgeQA
    Source      string         `json:"source"      gorm:"type:varchar(32);not null"`  // 来源: local / remote / openai / zhipu...
    Status      string         `json:"status"      gorm:"type:varchar(32);default:'active'"` // 状态: active / downloading / download_failed
    Parameters  ModelParameters `json:"parameters"  gorm:"type:json"`                 // 模型参数
    CreatedAt   time.Time      `json:"created_at"`
    UpdatedAt   time.Time      `json:"updated_at"`
    DeletedAt   gorm.DeletedAt `json:"deleted_at"   gorm:"index"`
}

// ModelParameters 模型参数：不同的ModelType用不同的字段
type ModelParameters struct {
    // Chat 参数
    BaseURL        string  `json:"base_url"`         // API endpoint
    APIKey         string  `json:"api_key"`          // API 密钥
    APIVersion     string  `json:"api_version"`      // API 版本
    ModelName      string  `json:"model_name"`       // 实际模型名（如 gpt-4o）
    Temperature    float64 `json:"temperature"`       // 温度参数
    MaxTokens      int     `json:"max_tokens"`        // 最大输出Token
    // Embedding 参数
    Dimension      int     `json:"dimension"`         // 向量维度
    // Ollama 参数
    KeepAlive      string  `json:"keep_alive"`        // Ollama 保活时间
}

// ModelType 模型类型枚举
const (
    ModelTypeEmbedding   = "Embedding"    // Embedding模型
    ModelTypeRerank      = "Rerank"       // Rerank模型
    ModelTypeKnowledgeQA = "KnowledgeQA"  // 对话模型（用于知识问答）
)
```

### Chat — 对话请求/响应

```go
// ChatRequest 对话请求
type ChatRequest struct {
    Query              string            `json:"query"`                // 用户问题
    SessionID          string            `json:"session_id"`           // 会话ID
    KnowledgeBaseIDs   []string          `json:"knowledge_base_ids"`   // 指定搜索的知识库ID列表
    Images             []string          `json:"images"`               // 附带图片URL
    MentionedItems     []MentionedItem   `json:"mentioned_items"`      // @提及的知识库/文件
}

// ChatResponse 对话响应（非流式）
type ChatResponse struct {
    Content     string       `json:"content"`       // 回复正文
    References  []Reference  `json:"references"`    // 引用来源
    TokenUsage  TokenUsage   `json:"usage"`         // Token用量
}

// TokenUsage Token用量统计
type TokenUsage struct {
    PromptTokens     int `json:"prompt_tokens"`
    CompletionTokens int `json:"completion_tokens"`
    TotalTokens      int `json:"total_tokens"`
}

// Reference 引用来源（检索到的chunk）
type Reference struct {
    ChunkID        string  `json:"chunk_id"`        // ChunkID
    Content        string  `json:"content"`          // Chunk正文
    Score          float64 `json:"score"`            // 相似度分数
    KnowledgeID    string  `json:"knowledge_id"`     // 来源知识条目ID
    KnowledgeTitle string  `json:"knowledge_title"`  // 来源知识条目标题
}
```
